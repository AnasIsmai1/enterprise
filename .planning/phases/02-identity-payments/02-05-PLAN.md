    export class PremiumGuard implements CanActivate {

        const { user } = context.switchToHttp().getRequest();
        if (!user) return false;
        // Premium and superadmin users pass; free users get 403

    **Step 3: Add view-only enforcement to PetService**

    LOCKED DECISIONS:
    - "View-only" means: can view all pet data but cannot edit, add records, or complete tasks
    - User chooses which pet remains fully active on downgrade

    Add a helper method to PetService:
    ```typescript
    async isViewOnly(userId: string, petId: string): Promise<boolean> {
      const user = await this.userRepo.findOne({ where: { id: userId } });
      if (!user) return true;
      // Premium and superadmin users: no view-only restriction
      if (user.role === UserRole.PREMIUM || user.role === UserRole.SUPERADMIN) return false;
      // Free users: only the active pet is fully accessible
      return user.activePetId !== petId;
    }
    ```

    Update PetService mutation methods (update, softDelete) to check view-only:
    ```typescript
    async update(userId: string, petId: string, dto: UpdatePetDto): Promise<Pet> {
      const viewOnly = await this.isViewOnly(userId, petId);
      if (viewOnly) {
        throw new ForbiddenException(
          'This pet is in view-only mode. Upgrade to premium or set this pet as your active pet.'
        );
      }
      // ... existing update logic
    }
    ```
    Apply the same check to `softDelete` and any future mutation endpoints.

    `findOne` and `findAllByUser` should include a `viewOnly` boolean in the response so the frontend can show the appropriate UI:

    Update `handleSubscriptionDeleted`:
    ```typescript
    private async handleSubscriptionDeleted(sub: Stripe.Subscription): Promise<void> {
      // Find subscription by stripeSubscriptionId
      const subscription = await this.subscriptionRepo.findOne({
        where: { stripeSubscriptionId: sub.id },
      });
      if (!subscription) return;

      // Update subscription status
      subscription.status = SubscriptionStatus.CANCELED;
      await this.subscriptionRepo.save(subscription);

      // Downgrade user role to FREE
      const user = await this.userRepo.findOne({ where: { id: subscription.userId } });
      if (!user) return;
      user.role = UserRole.FREE;
      await this.userRepo.save(user);

      // SUB-12: Data is preserved. View-only enforcement is handled by PetService.isViewOnly()
      // at request time -- no need to modify pet records. The user's current activePetId
      // becomes their "primary" pet, and all others are view-only.

      // Audit log: user downgraded
      await this.auditLogService.log({
        userId: subscription.userId,
        action: 'SUBSCRIPTION_DOWNGRADE',
        details: { subscriptionId: sub.id, previousPlan: subscription.plan },
      });
    }
    ```
